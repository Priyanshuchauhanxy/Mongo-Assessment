// Task 3: Find Pop or Hip-Hop playlists using $in and sort by likes descending

db.playlists.find(
    {
        genre: { $in: ["Pop", "Hip-Hop"] }
    }
).sort({
    likes: -1
});

// Task 4: Likes between 500 and 2000 (inclusive), sort by followers,return top 3

db.playlists.find(
    {
        likes: {
            $gte: 500,
            $lte: 2000
        }
    }
)
.sort({
    followers: -1
})
.limit(3);

// Task 5: Trending Playlists More than 5000 followers AND at least 1000 likes
